9pub(crate) fn build_mod_item_ids(state: &ModState) -> Tokens {
10    quote! {
11        public class ModItemIds {
12            private static $(resource_key())<$(item())> create(String name) {
13                $(identifier()) id = $(identifier()).fromNamespaceAndPath($(&state.mod_name).MOD_ID, name);
14                return $(resource_key()).create($(registries()).ITEM, id);
15            }
16
17            $("// Mod Item ID Registration")
18            $(for i in &state.items =>
19                public static final $(resource_key())<$(item())> $(to_upper(&i.id)) = create(
20                    $(quoted(&i.id))
21                );
22            )
23        }

27pub(crate) fn build_mod_items(state: &ModState) -> Tokens {
28    quote! {
29        public class ModItems {
30            private static $(item()) register($(resource_key())<$(item())> itemKey, $(function())<$(item()).Properties, $(item())> itemFactory, $(item()).Properties settings) {
31                $(item()) item = itemFactory.apply(settings.setId(itemKey));
32                $(registry()).register($(built_in_registries()).ITEM, itemKey, item);
33                return item;
34            }
35
36            $("// Item Registration")
37            $(for i in &state.items =>
38                public static final $(item()) $(to_upper(&i.id)) = register(ModItemIds.$(to_upper(&i.id)), $(item())::new, $(item_properties(i)));$['\r']
39            )
40
41            public static void initialize() {
42                $(if !&state.items.is_empty() =>
43                    $(creative_mode_tab_events()).modifyOutputEvent($(creative_mode_tabs()).INGREDIENTS)
44                        .register((creativeTab) ->
45                        {
46                            $(for i in &state.items =>
47                                creativeTab.accept($(mod_items(state)).$(to_upper(&i.id)));$['\r']
48                            )
49                        });
50                )
51            }
52        }
53    }

56pub(crate) fn build_mod_blocks(state: &ModState) -> Tokens {
57    quote! {
58        public class ModBlocks {
59            private static $(block()) register($(block_item_id()) id, $(function())<$(block_behaviour()).Properties, $(block())> blockFactory, $(block_behaviour()).Properties properties) {
60                $(block()) block = register(id.block(), blockFactory, properties);
61                $(block_item()) blockItem = new $(block_item())(block, new $(item()).Properties().useBlockDescriptionPrefix().setId(id.item()));
62                $(registry()).register($(built_in_registries()).ITEM, id.item(), blockItem);
63                return block;
64            }
65
66            private static $(block()) register($(resource_key())<$(block())> id, $(function())<$(block_behaviour()).Properties, $(block())> blockFactory, $(block_behaviour()).Properties properties) {
67                $(block()) block = blockFactory.apply(properties.setId(id));
68                return $(registry()).register($(built_in_registries()).BLOCK, id, block);
69            }
70
71            $("// Block Registration")
72            $(for b in &state.blocks =>
73                public static final $(block()) $(to_upper(&b.id)) = register(
74                    ModBlockItemIds.$(to_upper(&b.id)),
75                    $(block())::new,
76                    $(block_behaviour()).Properties.ofFullCopy($(blocks()).DIRT)
77                );$['\r']
78            )
79
80            public static void initialize() {
81                $(if !&state.blocks.is_empty() =>
82                    $(creative_mode_tab_events()).modifyOutputEvent($(creative_mode_tabs()).INGREDIENTS)
83                        .register((creativeTab) ->
84                        {
85                            $(for b in &state.blocks =>
86                                creativeTab.accept($(mod_blocks(state)).$(to_upper(&b.id)));$['\r']
87                            )
88                        });
89                )
90            }
91        }
92    }

242fn recipe_call(recipe: &Recipe, state: &ModState) -> Tokens {
243    let result_ref = result_item(recipe, state);
244    let count = recipe.count;
245    let recipe_id = strip_namespace(&recipe.id);
246    let full_id = format!("{}:{}", state.mod_id, recipe_id);
247    match recipe.kind.as_str() {
248        "crafting_shaped" => {
249            let pattern_lines = &recipe.pattern;
250            let defines = &recipe.ingredients;
251            let result_ref2 = result_ref.clone();
252            let result_ref3 = result_ref.clone();
253            quote! {
254                shaped($(recipe_category()).MISC, $(result_ref), $(count))
255                    $(for p in pattern_lines => .pattern($(quoted(p))))
256                    $(for (k, v) in defines => .define($(format!("'{}'", k.chars().next().unwrap_or('?'))), $(ingredient_ref(v, state))))
257                    .unlockedBy(getHasName($(result_ref2)), has($(result_ref3)))
258                    .save(exporter, $(quoted(&full_id)));
259            }
260        }
261        "crafting_shapeless" => {
262            let ingredients = &recipe.ingredients;
263            let result_ref2 = result_ref.clone();
264            let result_ref3 = result_ref.clone();
265            quote! {
266                shapeless($(recipe_category()).MISC, $(result_ref), $(count))
267                    $(for (_, v) in ingredients => .requires($(ingredient_ref(v, state))))
268                    .unlockedBy(getHasName($(result_ref2)), has($(result_ref3)))
269                    .save(exporter, $(quoted(&full_id)));
270            }
271        }
272        _ => quote! { /* unsupported recipe type: $(quoted(&recipe.kind)) */ },
273    }
274}

276fn result_item(recipe: &Recipe, state: &ModState) -> Tokens {
277    let id = &recipe.result;
278    if let Some(vanilla) = id.strip_prefix("minecraft:") {
279        let mc_constant = vanilla.to_uppercase().replace('-', "_");
280        quote! { $(items()).$(mc_constant) }
281    } else {
282        let item_id = strip_namespace(id);
283        quote! { $(mod_items(state)).$(to_upper(item_id)) }
284    }
285}

287fn ingredient_ref(value: &str, state: &ModState) -> Tokens {
288    if let Some(vanilla) = value.strip_prefix("minecraft:") {
289        let mc_constant = vanilla.to_uppercase().replace('-', "_");
290        quote! { $(ingredient()).of($(items()).$(mc_constant)) }
291    } else {
292        let item_id = strip_namespace(value);
293        let const_name = to_upper(item_id);
294        if state.items.iter().any(|i| i.id == item_id) {
295            quote! { $(ingredient()).of($(mod_items(state)).$(const_name)) }
296        } else if state.blocks.iter().any(|b| b.id == item_id) {
297            quote! { $(ingredient()).of($(mod_blocks(state)).$(const_name)) }
298        } else {
299            let id_ref: Tokens = quote! {
300                $(identifier()).fromNamespaceAndPath(
301                    $(mod_class(state)).MOD_ID,
302                    $(quoted(item_id))
303                )
304            };
305            quote! { $(ingredient()).of($id_ref) }
306        }
307    }
308}

310/// Strips the `namespace:` prefix from an item identifier, returning the bare id.
311fn strip_namespace(id: &str) -> &str {
312    id.split_once(':').map_or(id, |(_, path)| path)
313}

315fn item_properties(item: &Item) -> Tokens {
316    let mut out: Tokens = quote! { new Item.Properties() };
317
318    if item.kind == ItemKind::Tool
319        && let Some(mat) = &item.material
320    {
321        let material_const = mat.to_uppercase().replace('-', "_");
322        let damage = format!("{}f", item.attack_damage.unwrap_or(1.0));
323        let speed = format!("{}f", item.attack_speed.unwrap_or(1.6));
324        quote_in! { out => .sword($(tool_materials()).$(material_const), $(damage), $(speed)) }
325    }
326
327    if let Some(dur) = item.durability {
328        let snippet = format!("durability({})", dur);
329        quote_in! { out => .$(snippet) }
330    }
331
332    out
333}
