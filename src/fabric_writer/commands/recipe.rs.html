6/// Adds a recipe to the mod state and regenerates Java sources.
7pub fn add(args: RecipeAddArgs) -> Result<()> {
8    let verbose = args.verbose;
9    let mut state = state::load()?;
10    let recipe = build_recipe(&args)?;
11    let entity = Entity::Recipe(recipe.clone());
12    let recipe_label = format!("{} ({})", recipe.id, recipe.kind);
13    let dirty = DirtyFlags::from_entity(&entity);
14    state.add(entity)?;
15    state.save()?;
16    regenerate_all(&state, dirty, verbose)?;
17    println!("Added recipe: {}", recipe_label);
18    Ok(())
19}
20
21fn build_recipe(args: &RecipeAddArgs) -> Result<Recipe> {
22    let mut recipe = Recipe::new(&args.id)?;
23    if let Some(kind) = &args.kind {
24        recipe.kind = kind.clone();
25    }
26    recipe.result = args.result.clone().unwrap_or_default();
27    recipe.count = args.count.unwrap_or(1);
28    recipe.pattern = args.pattern.clone();
29    for pair in &args.ingredients {
30        let (key, value) = parse_ingredient_pair(pair)?;
31        recipe.ingredients.insert(key, value);
32    }
33    Ok(recipe)
34}
35
36fn parse_ingredient_pair(pair: &str) -> Result<(String, String)> {
37    let Some(idx) = pair.find('=') else {
38        bail!("Ingredient pair '{}' must be in key=value format", pair);
39    };
40    let key = pair[..idx].trim().to_string();
41    let value = pair[idx + 1..].trim().to_string();
42    if key.is_empty() || value.is_empty() {
43        bail!("Ingredient pair '{}' has an empty key or value", pair);
44    }
45    Ok((key, value))
46}

48/// Removes a recipe from the mod state and regenerates Java sources.
49pub fn remove(args: RecipeRemoveArgs) -> Result<()> {
50    let verbose = args.verbose;
51    let mut state = state::load()?;
52    let entity = Entity::Recipe(Recipe::new(&args.id)?);
53    let dirty = DirtyFlags::from_entity(&entity);
54    state.remove(entity)?;
55    state.save()?;
56    regenerate_all(&state, dirty, verbose)?;
57    println!("Removed recipe: {}", args.id);
58    Ok(())
59}

61/// CLI arguments for `fw add recipe`.
62#[derive(Parser)]
63#[command(after_help = "\
64Examples:
65  # Shaped recipe: 3 sticks in a T-pattern -> 1 iron_sword
66  fw add recipe craft_iron_sword --kind crafting_shaped --result iron_sword \\
67      --pattern \"S S\" --pattern \" S \" --pattern \" S \" \\
68      --ingredients S=minecraft:stick
69
70  # Shapeless recipe: copper_ingot + diamond -> 1 enchanted_apple
71  fw add recipe enchanted_apple --kind crafting_shapeless --result minecraft:enchanted_golden_apple \\
72      --ingredients C=copper_ingot --ingredients D=minecraft:diamond
73
74  # Vanilla result with multiple ingredient types
75  fw add recipe wood_apple --kind crafting_shaped --result minecraft:apple \\
76      --pattern \"WWW\" --pattern \" S \" --pattern \" S \" \\
77      --ingredients W=minecraft:logs --ingredients S=minecraft:stick
78
79  Pattern notes:
80  - Each --pattern line is one row of the crafting grid
81  - Each unique character in the pattern is an ingredient key
82  - --ingredients maps key=value where value is minecraft:<id> or <modid>:<id>")]
83pub struct RecipeAddArgs {
84    /// Recipe ID.
85    pub id: String,
86
87    /// Recipe type (e.g. `crafting_shaped`).
88    #[arg(long)]
89    pub kind: Option<String>,
90
91    /// Result item ID.
92    #[arg(long)]
93    pub result: Option<String>,
94
95    /// Number of items produced.
96    #[arg(long)]
97    pub count: Option<u32>,
98
99    /// Crafting pattern lines (e.g. `--pattern "WWW"` repeated).
100    #[arg(long)]
101    pub pattern: Vec<String>,
102
103    /// Ingredient key→value mapping (e.g. `--ingredients W=minecraft:logs`).
104    #[arg(long)]
105    pub ingredients: Vec<String>,
106
107    /// Show which files were regenerated, skipped, or pruned
108    #[arg(short = 'v', long, default_value_t = false)]
109    pub verbose: bool,
110}
